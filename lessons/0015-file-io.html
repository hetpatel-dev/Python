Lesson 15: File I/O — Reading, Writing, and Working with Data

concept File I/O

Programs need to read data from files and write results back. Python's file I/O toolkit — open(), with statement, pathlib, csv, json, os, and shutil — provides everything from low-level file operations to high-level directory management. This lesson covers the full file I/O ecosystem from the official Python tutorial on input and output.

Primary sources: Tutorial — Input and Output, io — Core tools for working with streams, pathlib — Object-oriented filesystem paths, csv — CSV File Reading and Writing, json — JSON encoder and decoder, os — Miscellaneous operating system interfaces, shutil — High-level file operations.

1. Opening and Closing Files

The built-in open() function returns a file object. It's most commonly used with two positional arguments and one keyword argument: open(filename, mode='r', encoding=None).

File Modes

Mode Description File Exists? File Missing?
"r" Read (default) Read only FileNotFoundError
"w" Write (overwrites!) Erases content Creates new file
"a" Append Adds to end Creates new file
"x" Exclusive create FileExistsError Creates new file
"r+" Read + Write Both read and write FileNotFoundError
"rb", "wb" Binary mode Read/write bytes Varies by mode

Text vs Binary Mode

# Text mode (default) — strings, encoding/decoding
with open("data.txt", "r", encoding="utf-8") as f:
    text = f.read()  # Returns str

# Binary mode — bytes objects, no encoding
with open("image.jpg", "rb") as f:
    data = f.read()  # Returns bytes

Why Always Use 'with'

# BAD — manual close is risky
f = open("data.txt")
content = f.read()
# What if an exception occurs here?
f.close()  # Might never execute!

# GOOD — context manager guarantees closure
with open("data.txt", encoding="utf-8") as f:
    content = f.read()
# f.close() is called automatically, even if exception occurs

# Verify closure
with open("data.txt") as f:
    content = f.read()
print(f.closed)  # True
💡 Always specify encoding: Use encoding="utf-8" explicitly. The default encoding is platform-dependent (locale encoding). Python 3.15+ will make UTF-8 the default (PEP 686), but specify it now for compatibility.

2. Reading Files

File objects provide multiple ways to read content. Choose based on file size and processing needs.

with open("data.txt", encoding="utf-8") as f:
    # Read entire file at once (good for small files)
    content = f.read()           # Returns entire file as str
    content = f.read(100)        # Read first 100 characters

    # Reset position to beginning
    f.seek(0)

    # Read one line at a time
    line = f.readline()          # Includes trailing \n
    line = f.readline(50)        # Read at most 50 characters from current line

    # Reset position again
    f.seek(0)

    # Read all lines into a list
    lines = f.readlines()        # Returns list of strings

    # Reset position again
    f.seek(0)

    # Most memory-efficient: iterate directly
    for line in f:
        process(line.strip())    # strip() removes trailing \n

Reading Large Files

# BAD — loads entire file into memory
with open("huge.log", encoding="utf-8") as f:
    content = f.read()  # Could use GB of memory!

# GOOD — iterate line by line
with open("huge.log", encoding="utf-8") as f:
    for line in f:  # Only one line in memory at a time
        process(line)

# GOOD — read in chunks
with open("huge.log", "rb") as f:
    while True:
        chunk = f.read(8192)  # Read 8KB at a time
        if not chunk:
            break
        process(chunk)
💡 EOF detection: An empty string '' from read() or readline() means end of file. A blank line returns '\n', not an empty string.

3. Writing Files

# Write text (returns number of characters written)
with open("output.txt", "w", encoding="utf-8") as f:
    n = f.write("Hello, World!\n")  # Returns 14
    f.write("Second line\n")

# Append to existing file
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("New log entry\n")

# Write multiple lines (no newlines added automatically!)
lines = ["Line 1\n", "Line 2\n", "Line 3\n"]
with open("output.txt", "w", encoding="utf-8") as f:
    f.writelines(lines)

# Write bytes
with open("data.bin", "wb") as f:
    f.write(b"\x00\x01\x02\x03")

Buffering and Flushing

# Data is buffered for efficiency
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("data")
    # Data may not be written to disk yet!
    f.flush()  # Force buffer to disk

    # With statement calls flush() automatically on exit

4. File Position (tell and seek)

with open("data.txt", encoding="utf-8") as f:
    print(f.tell())    # 0 — current position

    f.read(10)         # Read first 10 characters
    print(f.tell())    # 10

    f.seek(0)          # Go back to beginning
    print(f.tell())    # 0

    f.seek(0, 2)       # Go to end (whence=2)
    print(f.tell())    # File size (in bytes for binary, varies for text)

    # Binary mode — full random access
with open("data.bin", "rb") as f:
    f.seek(100)        # Jump to byte 100
    data = f.read(50)  # Read 50 bytes from position 100
⚠️ Text mode seek limitations: In text mode, only seek(0) (beginning) and seek(0, 2) (end) are guaranteed to work. Other seeks may cause undefined behavior due to multi-byte encodings.

5. Encoding and Newlines

# Always specify encoding
with open("data.txt", encoding="utf-8") as f:
    content = f.read()

# Universal newlines (default)
with open("data.txt", newline=None) as f:  # Default behavior
    # \r\n, \r, \n all become \n
    content = f.read()

# No newline translation
with open("data.txt", newline='') as f:
    # Line endings preserved as-is
    content = f.read()

# On Windows, writing with \n converts to \r\n
# On Unix, \n stays as \n
with open("output.txt", "w", newline='', encoding="utf-8") as f:
    f.write("Line 1\nLine 2\n")  # \n written as-is
💡 CSV files require newline='': When reading/writing CSV files with the csv module, always use newline='' to prevent double newlines on Windows.

6. pathlib — Modern File Paths

The pathlib module (Python 3.4+) provides an object-oriented approach to file paths. Prefer pathlib over os.path in new code.

from pathlib import Path

# Create Path objects
p = Path("data/file.txt")           # Relative path
p = Path.home() / "documents"       # Home directory
p = Path.cwd()                      # Current working directory
p = Path("/usr/bin/python3")        # Absolute path

# The / operator joins paths
config = Path("project") / "config" / "settings.json"
# Path("project/config/settings.json")

# Path properties
p = Path("/home/user/data/report.csv")
print(p.name)       # "report.csv" — final component
print(p.stem)       # "report" — without suffix
print(p.suffix)     # ".csv" — file extension
print(p.parent)     # Path("/home/user/data") — parent directory
print(p.parts)      # ('/', 'home', 'user', 'data', 'report.csv')

Reading and Writing with pathlib

# Read text (convenience methods)
p = Path("data.txt")
content = p.read_text(encoding="utf-8")

# Write text
p.write_text("Hello, World!\n", encoding="utf-8")

# Read/write bytes
data = p.read_bytes()
p.write_bytes(b"\x00\x01\x02")

Path Operations

p = Path("project/data")

# Query
print(p.exists())          # True/False
print(p.is_file())         # True if file
print(p.is_dir())          # True if directory
print(p.is_symlink())      # True if symbolic link

# Metadata
stats = p.stat()
print(stats.st_size)       # Size in bytes
print(stats.st_mtime)      # Modification time

# Create directories
p.mkdir(parents=True, exist_ok=True)  # Create path and parents

# Delete file
p.unlink()                 # Delete file (not directory!)

# Rename/move
p.rename("new_name.txt")

Iterating and Globbing

# List directory contents
for p in Path("data").iterdir():
    print(p)

# Find files by pattern
for p in Path("project").glob("*.py"):
    print(p)  # All .py files in project/

# Recursive search
for p in Path("project").rglob("*.py"):
    print(p)  # All .py files in project/ and subdirectories
💡 pathlib vs os.path: pathlib is more readable (p.name vs os.path.basename(p)), more powerful (glob built-in), and more Pythonic. Use pathlib for new code.

7. os Module — File System Operations

import os

# Working directory
print(os.getcwd())         # Current directory
os.chdir("/tmp")           # Change directory

# Path operations
os.path.join("dir", "file.txt")     # "dir/file.txt"
os.path.basename("/a/b/file.txt")   # "file.txt"
os.path.dirname("/a/b/file.txt")    # "/a/b"
os.path.split("/a/b/file.txt")      # ("/a/b", "file.txt")
os.path.splitext("file.txt")        # ("file", ".txt")
os.path.abspath("relative")         # Absolute path
os.path.exists("path")              # True/False

# Directory operations
os.mkdir("new_dir")                 # Create directory
os.makedirs("a/b/c", exist_ok=True)  # Create nested directories
os.rmdir("empty_dir")               # Remove empty directory
os.listdir(".")                     # List directory contents

8. Directory Traversal

import os

# os.listdir — simple but no metadata
for name in os.listdir("data"):
    print(name)  # Just filenames

# os.scandir — efficient, returns DirEntry with cached metadata
with os.scandir("data") as entries:
    for entry in entries:
        print(entry.name)
        print(entry.is_dir())    # Cached, no extra stat() call
        print(entry.stat().st_size)  # File size

# os.walk — recursive traversal
for dirpath, dirnames, filenames in os.walk("project"):
    print(f"Directory: {dirpath}")
    for filename in filenames:
        print(f"  File: {filename}")

# os.walk bottom-up (for deleting empty directories)
for dirpath, dirnames, filenames in os.walk("project", topdown=False):
    os.rmdir(dirpath)  # Remove after contents are processed

# glob.glob — pattern matching
import glob
for p in glob.glob("**/*.py", recursive=True):
    print(p)  # All .py files recursively
💡 Use os.scandir(): os.scandir() is 2-20x faster than os.listdir() + os.path.isdir() because it caches metadata from the OS. Use it when you need to check file type or size.

9. File Metadata (os.stat)

import os
import stat

info = os.stat("file.txt")

# Size
print(info.st_size)        # Size in bytes

# Timestamps
print(info.st_mtime)       # Modification time (seconds since epoch)
print(info.st_atime)       # Access time
print(info.st_ctime)       # Creation (Windows) or metadata change (Unix)
print(info.st_mtime_ns)    # Nanosecond precision

# Convert to readable format
from datetime import datetime
mod_time = datetime.fromtimestamp(info.st_mtime)
print(mod_time.strftime("%Y-%m-%d %H:%M:%S"))

# File type from st_mode
if stat.S_ISREG(info.st_mode):
    print("Regular file")
elif stat.S_ISDIR(info.st_mode):
    print("Directory")
elif stat.S_ISLNK(info.st_mode):
    print("Symbolic link")

# Permission bits (Unix-style)
print(oct(info.st_mode & 0o777))  # e.g., 0o644

10. shutil — High-level File Operations

import shutil

# Copy files
shutil.copyfile("source.txt", "dest.txt")    # Copy content only
shutil.copy("source.txt", "dest.txt")        # Copy content + permissions
shutil.copy2("source.txt", "dest.txt")       # Copy content + permissions + metadata

# Copy directory tree
shutil.copytree("src_dir", "dst_dir")
shutil.copytree("src", "dst", dirs_exist_ok=True)  # Merge into existing
shutil.copytree("src", "dst", ignore=shutil.ignore_patterns("*.pyc", "__pycache__"))

# Move files/directories
shutil.move("file.txt", "new_location/")      # Move file
shutil.move("dir", "parent/")                 # Move directory

# Delete directory tree (use with caution!)
shutil.rmtree("directory_to_delete")
shutil.rmtree("dir", ignore_errors=True)      # Ignore errors

# Disk usage
usage = shutil.disk_usage("/")
print(f"Total: {usage.total // (1024**3)} GB")
print(f"Used: {usage.used // (1024**3)} GB")
print(f"Free: {usage.free // (1024**3)} GB")
⚠️ shutil.rmtree is dangerous: It recursively deletes everything! Always double-check the path. Consider using shutil.copytree() to backup first.

11. Temporary Files

import tempfile

# Named temporary file (has a name attribute)
with tempfile.NamedTemporaryFile(mode='w', suffix='.txt', delete=True) as f:
    f.write("temp data")
    print(f.name)  # e.g., /tmp/tmp12345.txt
# File deleted automatically when exiting with block

# Named temp file that persists after close
with tempfile.NamedTemporaryFile(delete=False) as f:
    f.write(b"persistent data")
    name = f.name
# File still exists at 'name' — must delete manually

# Temporary directory
with tempfile.TemporaryDirectory() as tmpdir:
    print(tmpdir)  # e.g., /tmp/tmp12345
    # Create files in tmpdir
    (Path(tmpdir) / "data.txt").write_text("hello")
# Directory and contents deleted automatically

# SpooledTemporaryFile — in-memory until threshold
with tempfile.SpooledTemporaryFile(max_size=1024 * 1024) as f:
    f.write(b"data")
    # Stays in memory until > 1MB, then rolls to disk

12. In-Memory Streams (StringIO, BytesIO)

import io

# StringIO — in-memory text stream
buffer = io.StringIO()
buffer.write("Hello ")
buffer.write("World!")
print(buffer.getvalue())  # "Hello World!"
buffer.seek(0)
print(buffer.read())      # "Hello World!"
buffer.close()

# Initialize with content
buffer = io.StringIO("Initial content")
print(buffer.read())      # "Initial content"

# BytesIO — in-memory binary stream
buffer = io.BytesIO()
buffer.write(b"Hello ")
buffer.write(b"World!")
print(buffer.getvalue())  # b"Hello World!"
buffer.close()

# Use cases: testing, building content before writing, CSV in memory
# Pass StringIO to csv module instead of file
import csv
output = io.StringIO()
writer = csv.writer(output)
writer.writerow(["Name", "Age"])
writer.writerow(["Alice", 30])
csv_content = output.getvalue()  # "Name,Age\nAlice,30\n"

13. Working with CSV Files

import csv

# Reading CSV
with open("data.csv", newline='', encoding="utf-8") as f:
    reader = csv.reader(f)
    header = next(reader)  # First row as header
    for row in reader:
        print(row)  # Each row is a list of strings

# Reading CSV as dictionaries
with open("data.csv", newline='', encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row["Name"], row["Age"])  # Access by column name

# Writing CSV
with open("output.csv", "w", newline='', encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["Name", "Age"])  # Write header
    writer.writerow(["Alice", 30])
    writer.writerow(["Bob", 25])
    writer.writerows([["Charlie", 35], ["Diana", 28]])

# Writing CSV from dictionaries
with open("output.csv", "w", newline='', encoding="utf-8") as f:
    fieldnames = ["Name", "Age", "City"]
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerow({"Name": "Alice", "Age": 30, "City": "NYC"})
    writer.writerow({"Name": "Bob", "Age": 25, "City": "LA"})

# Custom delimiter
with open("data.tsv", "w", newline='', encoding="utf-8") as f:
    writer = csv.writer(f, delimiter='\t')
    writer.writerow(["Name", "Age"])
💡 Why newline=''? On Windows, Python's default newline handling converts \n to \r\n when writing. The csv module adds its own newlines. Without newline='', you get double newlines: \r\r\n.

14. Working with JSON Files

import json

# Python to JSON type conversion:
# dict       → object
# list/tuple → array
# str        → string
# int/float  → number
# True       → true
# False      → false
# None       → null

data = {
    "name": "Alice",
    "age": 30,
    "scores": [95, 87, 92],
    "active": True,
    "address": None
}

# Write JSON to file
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, indent=2, ensure_ascii=False)

# Read JSON from file
with open("data.json", encoding="utf-8") as f:
    loaded = json.load(f)

# Serialize to string (with pretty printing)
json_str = json.dumps(data, indent=2, ensure_ascii=False)
print(json_str)
# {
#   "name": "Alice",
#   "age": 30,
#   "scores": [95, 87, 92],
#   "active": true,
#   "address": null
# }

# Deserialize from string
obj = json.loads(json_str)

# Custom default for non-serializable types
import datetime
def default_handler(obj):
    if isinstance(obj, datetime.datetime):
        return obj.isoformat()
    raise TypeError(f"Object of type {type(obj)} is not JSON serializable")

json.dumps({"timestamp": datetime.datetime.now()}, default=default_handler)

JSON Lines (one JSON object per line)

# Write JSON lines (useful for large datasets)
with open("data.jsonl", "w", encoding="utf-8") as f:
    for item in items:
        f.write(json.dumps(item) + "\n")

# Read JSON lines
with open("data.jsonl", encoding="utf-8") as f:
    items = [json.loads(line) for line in f]

15. Binary Files

# Reading/writing bytes
with open("data.bin", "wb") as f:
    f.write(b"\x00\x01\x02\x03\x04")

with open("data.bin", "rb") as f:
    data = f.read()  # b"\x00\x01\x02\x03\x04"

# struct module for packing/unpacking binary data
import struct
# Pack integers into bytes
data = struct.pack("iii", 1, 2, 3)  # 12 bytes
# Unpack bytes back to integers
a, b, c = struct.unpack("iii", data)  # a=1, b=2, c=3

16. Error Handling in File I/O

from pathlib import Path

# Common file I/O exceptions:
# FileNotFoundError — file doesn't exist
# PermissionError — no access rights
# FileExistsError — file already exists (x mode)
# IsADirectoryError — expected file, got directory
# NotADirectoryError — expected directory, got file
# OSError — base class for file system errors

# Graceful handling patterns
try:
    with open("config.txt", encoding="utf-8") as f:
        config = f.read()
except FileNotFoundError:
    print("Config file not found, using defaults")
    config = "{}"
except PermissionError:
    print("No permission to read config file")
    config = "{}"

# Check existence before operations
p = Path("data.txt")
if p.exists():
    content = p.read_text(encoding="utf-8")
else:
    print(f"File not found: {p}")

# Safe directory operations
import shutil
try:
    shutil.rmtree("directory")
except FileNotFoundError:
    pass  # Already deleted
except PermissionError as e:
    print(f"Cannot delete: {e}")

17. Best Practices Summary

# 1. Always use 'with' statement
with open("file.txt") as f:  # ✅
    content = f.read()

# 2. Always specify encoding
with open("file.txt", encoding="utf-8") as f:  # ✅
    content = f.read()

# 3. Don't load entire large files into memory
for line in open("huge.log"):  # ✅ iterate
    process(line)

# 4. Use pathlib for modern path handling
from pathlib import Path
p = Path("data") / "file.txt"  # ✅

# 5. Use os.scandir for directory listing with metadata
with os.scandir("dir") as entries:  # ✅ faster
    for entry in entries:
        print(entry.name, entry.stat().st_size)

# 6. CSV files: always use newline=''
with open("data.csv", newline='', encoding="utf-8") as f:  # ✅
    reader = csv.reader(f)

# 7. JSON: use ensure_ascii=False for Unicode
json.dumps(data, ensure_ascii=False)  # ✅

# 8. Use tempfile for temporary files
with tempfile.NamedTemporaryFile() as f:  # ✅ auto-cleanup
    f.write(b"data")

# 9. Binary mode for non-text files
with open("image.jpg", "rb") as f:  # ✅
    data = f.read()

# 10. Handle errors gracefully
try:
    with open("file.txt") as f:
        content = f.read()
except FileNotFoundError:
    content = default_content
